cmake_minimum_required(VERSION 3.20)
project(maskfish CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
option(MASKFISH_NATIVE "Tune for the build machine" ON)
option(MASKFISH_SANITIZE "Address/UB sanitizers" OFF)
option(MASKFISH_UNIVERSAL "One binary with a build per instruction set, the best one chosen at startup (Linux x86-64 and arm64)" OFF)
option(MASKFISH_EMBED_NET "Embed the network MASKFISH_NET_FILE in the binary (EvalFile falls back on it)" ON)
set(MASKFISH_NET_FILE "${CMAKE_SOURCE_DIR}/nets/maskfish.nnue" CACHE FILEPATH
    "The network embedded with MASKFISH_EMBED_NET (scripts/net.sh fetches the published one there)")

option(MASKFISH_LTO "Link-time optimisation (measured no faster)" OFF)
# The board as sets of squares beside the array (src/bitboard.h, from AB-JChess): the same moves in the same order
# and the same search node for node, found faster. A build option, so that the two builds can be tested against
# each other; every target is built with it (the layout of Position depends on it).
option(MASKFISH_BITBOARD "Move generation, legality, checks and the exchange evaluation on bitboards" OFF)
option(MASKFISH_PEXT "With MASKFISH_BITBOARD: gather a file's occupancy with PEXT (CPUs with a fast one)" OFF)
if(MASKFISH_BITBOARD)
    add_compile_definitions(MASKFISH_BITBOARD)
    if(MASKFISH_PEXT)
        add_compile_definitions(MASKFISH_PEXT)
    endif()
endif()
if(MASKFISH_LTO)  # before the targets: the variable sets their property as they are created
    include(CheckIPOSupported)
    check_ipo_supported(RESULT MASKFISH_HAS_IPO OUTPUT MASKFISH_IPO_MSG)
    if(MASKFISH_HAS_IPO)
        set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
    endif()
endif()

set(MASKFISH_CORE_SOURCES src/tables.cpp src/position.cpp src/eval.cpp src/nnue.cpp src/hnn/hnnue.cpp src/search.cpp)
if(MASKFISH_BITBOARD)
    list(APPEND MASKFISH_CORE_SOURCES src/bitboard.cpp)
endif()
set(MASKFISH_MAIN_SOURCES src/main.cpp src/uci.cpp src/selfplay.cpp)

# The embedded network (src/embedded.h): MASKFISH_NET_FILE, included in embedded_net.cpp with .incbin, when it
# exists and is a network file the engine reads (by its magic). Without it the binary embeds none.
set(MASKFISH_EMBEDDED_NET_CPP ${CMAKE_BINARY_DIR}/embedded_net.cpp)
set(MASKFISH_EMBEDDED "none")
if(MASKFISH_EMBED_NET)
    if(EXISTS "${MASKFISH_NET_FILE}")
        file(READ "${MASKFISH_NET_FILE}" magic LIMIT 8 HEX)
        # JQHNNUE1 (raw), JQHNNL01, JQHNNZ01 (compressed)
        if(magic MATCHES "^4a51484e4e554531$|^4a51484e4e4c3031$|^4a51484e4e5a3031$")
            set(MASKFISH_EMBEDDED "${MASKFISH_NET_FILE}")
        else()
            message(FATAL_ERROR "MASKFISH_NET_FILE ${MASKFISH_NET_FILE} is not a network file the engine reads")
        endif()
    else()
        message(WARNING "No network to embed: ${MASKFISH_NET_FILE} does not exist (sh scripts/net.sh fetches the "
                        "published one; then run cmake again). The binary embeds none.")
    endif()
endif()
message(STATUS "Embedded network: ${MASKFISH_EMBEDDED}")
if(MASKFISH_EMBEDDED STREQUAL "none")
    set(MASKFISH_EMBEDDED_BODY "EmbeddedNet embedded_net() { return {}; }")
else()
    set(MASKFISH_EMBEDDED_BODY
"__asm__(\".section .rodata.maskfish_net,\\\"a\\\"\\n.balign 64\\n\"
        \".globl maskfish_net_begin\\n.hidden maskfish_net_begin\\nmaskfish_net_begin:\\n\"
        \".incbin \\\"${MASKFISH_EMBEDDED}\\\"\\n\"
        \".globl maskfish_net_end\\n.hidden maskfish_net_end\\nmaskfish_net_end:\\n.previous\\n\");
extern \"C\" const uint8_t maskfish_net_begin[], maskfish_net_end[];
EmbeddedNet embedded_net() { return {maskfish_net_begin, std::size_t(maskfish_net_end - maskfish_net_begin)}; }")
endif()
file(CONFIGURE OUTPUT ${MASKFISH_EMBEDDED_NET_CPP} CONTENT
"// Generated by CMakeLists.txt (MASKFISH_EMBED_NET, MASKFISH_NET_FILE): the network embedded in the binary.
#include \"embedded.h\"

namespace jq {
${MASKFISH_EMBEDDED_BODY}
}  // namespace jq
" @ONLY)
if(NOT MASKFISH_EMBEDDED STREQUAL "none")
    set_source_files_properties(${MASKFISH_EMBEDDED_NET_CPP} PROPERTIES OBJECT_DEPENDS "${MASKFISH_EMBEDDED}")
endif()
add_library(maskfish_core STATIC ${MASKFISH_CORE_SOURCES})
target_include_directories(maskfish_core PUBLIC src)
# The integer kernels and the net decoder of the network evaluator are compiled without floating-point contraction.
set_source_files_properties(src/hnn/hnnue.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
# jqnz_tool: packs a raw .hnn net into the compressed .nnue form and back (tools/hnn/jqnz_tool.cpp).
add_executable(jqnz_tool EXCLUDE_FROM_ALL tools/hnn/jqnz_tool.cpp)
target_include_directories(jqnz_tool PRIVATE src/hnn)
target_compile_options(maskfish_core PRIVATE -Wall -Wextra)
find_package(Threads REQUIRED)
target_link_libraries(maskfish_core PUBLIC Threads::Threads)
if(MASKFISH_SANITIZE)
    target_compile_options(maskfish_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(maskfish_core PUBLIC -fsanitize=address,undefined)
endif()

if(NOT MASKFISH_UNIVERSAL)
    if(MASKFISH_NATIVE)
        target_compile_options(maskfish_core PUBLIC -march=native)
        set(MASKFISH_BUILD native)
    elseif(CMAKE_CXX_FLAGS MATCHES "-march=([^ ]+)")
        set(MASKFISH_BUILD ${CMAKE_MATCH_1})
    else()
        set(MASKFISH_BUILD portable)
    endif()
    add_executable(maskfish ${MASKFISH_MAIN_SOURCES} ${MASKFISH_EMBEDDED_NET_CPP})
    target_link_libraries(maskfish PRIVATE maskfish_core)
    target_compile_options(maskfish PRIVATE -Wall -Wextra)
    target_compile_definitions(maskfish PRIVATE MASKFISH_BUILD="${MASKFISH_BUILD}")
else()
    # The universal binary (src/universal.h): the engine built for the baseline of the architecture, with the engine
    # built for every higher instruction set (the "slices", lowest first) embedded in it as executables. Every build
    # has its own flags, so MASKFISH_NATIVE and a -march in CMAKE_CXX_FLAGS do not apply.
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "MASKFISH_UNIVERSAL is for Linux (x86-64 and arm64)")
    endif()
    if(MASKFISH_SANITIZE)
        message(FATAL_ERROR "MASKFISH_UNIVERSAL and MASKFISH_SANITIZE do not go together")
    endif()
    if(CMAKE_CXX_FLAGS MATCHES "-march=")
        message(FATAL_ERROR "MASKFISH_UNIVERSAL sets -march for every build: leave it out of CMAKE_CXX_FLAGS")
    endif()
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
        set(MASKFISH_BASELINE x86-64)
        set(MASKFISH_FLAGS_x86-64 -march=x86-64)
        # SSSE3 / SSE4.1 / POPCNT; AVX2 and BMI2; AVX-VNNI (Alder Lake); AVX-512 BW; AVX-512 with VNNI and VBMI2
        set(MASKFISH_SLICES x86-64-v2 x86-64-v3 x86-64-avxvnni x86-64-v4 x86-64-avx512icl)
        set(MASKFISH_FLAGS_x86-64-v2 -march=x86-64-v2)
        set(MASKFISH_FLAGS_x86-64-v3 -march=x86-64-v3)
        set(MASKFISH_FLAGS_x86-64-avxvnni -march=x86-64-v3 -mavxvnni)
        set(MASKFISH_FLAGS_x86-64-v4 -march=x86-64-v4)
        set(MASKFISH_FLAGS_x86-64-avx512icl -march=x86-64-v4 -mavx512vnni -mavx512vbmi2)
    elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
        set(MASKFISH_BASELINE armv8)
        set(MASKFISH_FLAGS_armv8 -march=armv8-a)
        set(MASKFISH_SLICES armv8-dotprod)  # the dot product instructions (ARMv8.2 and later)
        set(MASKFISH_FLAGS_armv8-dotprod -march=armv8.2-a+dotprod)
    else()
        message(FATAL_ERROR "MASKFISH_UNIVERSAL is for x86-64 and arm64, not ${CMAKE_SYSTEM_PROCESSOR}")
    endif()
    # The baseline: the binary itself (and the core of the differential test).
    target_compile_options(maskfish_core PUBLIC ${MASKFISH_FLAGS_${MASKFISH_BASELINE}})
    set(MASKFISH_SLICE_DIR ${CMAKE_BINARY_DIR}/universal)
    set(MASKFISH_SLICE_ASM "")
    set(MASKFISH_SLICE_DECLS "")
    set(MASKFISH_SLICE_TABLE "")
    set(MASKFISH_SLICE_FILES "")
    set(MASKFISH_SLICE_TARGETS "")
    foreach(slice IN LISTS MASKFISH_SLICES)
        string(MAKE_C_IDENTIFIER ${slice} id)
        add_library(maskfish_core_${id} STATIC EXCLUDE_FROM_ALL ${MASKFISH_CORE_SOURCES})
        target_include_directories(maskfish_core_${id} PUBLIC src)
        target_compile_options(maskfish_core_${id} PRIVATE -Wall -Wextra)
        target_compile_options(maskfish_core_${id} PUBLIC ${MASKFISH_FLAGS_${slice}})
        target_link_libraries(maskfish_core_${id} PUBLIC Threads::Threads)
        # The network embedded in the binary reaches the build through a memfd (src/embedded_fd.cpp)
        add_executable(maskfish_slice_${id} EXCLUDE_FROM_ALL ${MASKFISH_MAIN_SOURCES} src/embedded_fd.cpp)
        target_link_libraries(maskfish_slice_${id} PRIVATE maskfish_core_${id})
        target_compile_options(maskfish_slice_${id} PRIVATE -Wall -Wextra)
        target_compile_definitions(maskfish_slice_${id} PRIVATE MASKFISH_BUILD="${slice}" MASKFISH_UNIVERSAL_SLICE)
        target_link_options(maskfish_slice_${id} PRIVATE -s)
        set_target_properties(maskfish_slice_${id} PROPERTIES OUTPUT_NAME maskfish-${slice}
                              RUNTIME_OUTPUT_DIRECTORY ${MASKFISH_SLICE_DIR})
        set(file ${MASKFISH_SLICE_DIR}/maskfish-${slice})
        string(APPEND MASKFISH_SLICE_ASM
            "    \".balign 64\\n.globl maskfish_slice_${id}_begin\\n.hidden maskfish_slice_${id}_begin\\n\"\n"
            "    \"maskfish_slice_${id}_begin:\\n.incbin \\\"${file}\\\"\\n\"\n"
            "    \".globl maskfish_slice_${id}_end\\n.hidden maskfish_slice_${id}_end\\nmaskfish_slice_${id}_end:\\n\"\n")
        string(APPEND MASKFISH_SLICE_DECLS
            "extern \"C\" const unsigned char maskfish_slice_${id}_begin[], maskfish_slice_${id}_end[];\n")
        string(APPEND MASKFISH_SLICE_TABLE
            "    {\"${slice}\", maskfish_slice_${id}_begin, maskfish_slice_${id}_end},\n")
        list(APPEND MASKFISH_SLICE_FILES ${file})
        list(APPEND MASKFISH_SLICE_TARGETS maskfish_slice_${id})
    endforeach()
    list(LENGTH MASKFISH_SLICES MASKFISH_NUM_SLICES)
    set(MASKFISH_SLICES_CPP ${MASKFISH_SLICE_DIR}/universal_slices.cpp)
    file(CONFIGURE OUTPUT ${MASKFISH_SLICES_CPP} CONTENT
"// Generated by CMakeLists.txt (MASKFISH_UNIVERSAL): the builds embedded in the universal binary.
#include \"universal.h\"

__asm__(\".section .rodata.maskfish_slices,\\\"a\\\"\\n\"
${MASKFISH_SLICE_ASM}    \".previous\\n\");

${MASKFISH_SLICE_DECLS}
namespace jq::universal {
const Slice kSlices[] = {
${MASKFISH_SLICE_TABLE}};
const int kNumSlices = ${MASKFISH_NUM_SLICES};
const char* const kBaseline = \"${MASKFISH_BASELINE}\";
}  // namespace jq::universal
" @ONLY)
    set_source_files_properties(${MASKFISH_SLICES_CPP} PROPERTIES OBJECT_DEPENDS "${MASKFISH_SLICE_FILES}")
    add_executable(maskfish ${MASKFISH_MAIN_SOURCES} src/universal.cpp ${MASKFISH_SLICES_CPP} ${MASKFISH_EMBEDDED_NET_CPP})
    target_link_libraries(maskfish PRIVATE maskfish_core)
    target_compile_options(maskfish PRIVATE -Wall -Wextra)
    target_compile_definitions(maskfish PRIVATE MASKFISH_BUILD="${MASKFISH_BASELINE}" MASKFISH_UNIVERSAL)
    add_dependencies(maskfish ${MASKFISH_SLICE_TARGETS})
endif()

# The referee (tools/native/, an independent implementation of the rules): the static library
# jieqi_core for the differential test, and the shared library jieqi_rules (BUILD/lib) that the Python side
# (tools/jieqi/) loads for tools/match.py and tools/play.py.
option(JIEQI_SANITIZERS "Address/UB sanitizers for the referee" OFF)
add_subdirectory(tools/native)
add_executable(maskfish_difftest tests/difftest.cpp)
target_link_libraries(maskfish_difftest PRIVATE maskfish_core jieqi_core)
